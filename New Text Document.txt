import json
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import os

# Danh sách các file bạn đã cung cấp
file_names = [
    'scalars_cbam.json', 'scalars_cbam_aspp.json', 'scalars_aspp.json',
    'scalars_mask_r101.json', 'scalars_mask_r50.json', 'scalars_solo.json',
    'scalars_solov2.json', 'scalars_YOLACT.json', 'scalars_condinst.json'
]

def load_data(file_path):
    data = []
    with open(file_path, 'r') as f:
        for line in f:
            try:
                data.append(json.loads(line))
            except:
                continue
    return pd.DataFrame(data)

# Cấu hình biểu đồ
plt.figure(figsize=(15, 6))

# 1. Vẽ biểu đồ LOSS
plt.subplot(1, 2, 1)
for file in file_names:
    if os.path.exists(file):
        df = load_data(file)
        if 'loss' in df.columns:
            # Làm mượt đường cong (Optional)
            df['loss_smooth'] = df['loss'].rolling(window=10).mean()
            plt.plot(df['step'], df['loss_smooth'], label=file.replace('scalars_', '').replace('.json', ''))

plt.title('Training Loss Comparison')
plt.xlabel('Steps')
plt.ylabel('Loss')
plt.yscale('log') # Dùng log scale vì YOLACT loss rất cao so với các model khác
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)

# 2. Vẽ biểu đồ ACCURACY
plt.subplot(1, 2, 2)
for file in file_names:
    if os.path.exists(file):
        df = load_data(file)
        if 'acc' in df.columns:
            plt.plot(df['step'], df['acc'], label=file.replace('scalars_', '').replace('.json', ''))

plt.title('Training Accuracy Comparison')
plt.xlabel('Steps')
plt.ylabel('Accuracy (%)')
plt.legend()
plt.grid(True, linestyle='--', alpha=0.6)

plt.tight_layout()
plt.show()
